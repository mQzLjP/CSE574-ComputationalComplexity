\chapter{Proving NP-hard Problems}

\section{NP-hard and NP-complete}

\begin{definition}[Karp Reduction]
	$L \le_p L'$ if there's a polynomial-time computable function $f$ such that for all $x$, $x \in L \iff f(x) \in L'$.
\end{definition}

\begin{definition}[NP-hard and NP-complete]
	$L$ is \textbf{NP-hard} if for all $L' \in \NP$, $L' \le_p L$. $L$ is \textbf{NP-complete} if $L$ is NP-hard and $L \in \NP$.
\end{definition}

We can find a NP-complete problem by simply looking at the definitions.

\begin{problem}[TMSAT]
	\[\text{TMSAT} = \{\langle M, x, 1^n, 1^t\rangle : \exists u\in \{0,1\}^n \text{ such that } M(x,u) \text{ accepts in } t \text{ steps}\}.\]
\end{problem}
\begin{theorem}
	TMSAT is NP-complete.
\end{theorem}
\begin{proof}
	First of all we see that TMSAT is in NP as we can give out $u$ and simulate.

	Now we want to prove that TMSAT is NP-hard. Let $L \in \NP$ and let $M$ be a polynomial-time verifier for $L$. Let $p(n)$ be the polynomial bound on the running time of $M$. We can reduce $L$ to TMSAT by mapping $x$ to $\langle M, x, 1^{p(|x|)}, 1^{p(|x|)}\rangle$. This reduction is clearly polynomial-time computable.
	Now we have that $x \in L$ if and only if there exists a $u$ such that $M(x,u)$ accepts in $p(|x|)$ steps, which is equivalent to $\langle M, x, 1^{p(|x|)}, 1^{p(|x|)}\rangle \in \text{TMSAT}$. Therefore, TMSAT is NP-hard.
\end{proof}

This, however, does not give us a very useful NP-complete problem.

\section{Cook-Levin Theorem}

Cook-Levin theorem is about an interesting NP-complete problem called SAT.

\begin{definition}[CNF and SAT]
	A formula is in \textbf{conjunctive normal form} (CNF) if it is a Boolean formula over variables $u_1, u_2, \ldots, u_n$ that has the form
	\[ \bigwedge_i\left( \bigvee_j v_{ij}\right)\]
	where each $v_{ij}$ is either a variable $u_k$ or its negation $\neg u_k$ (also written as $\overline{u_k}$). A formula is \textbf{satisfiable} if there exists an assignment of truth values to the variables that makes the formula evaluate to true. The \textbf{SAT} problem is the decision problem of determining whether a given CNF formula is satisfiable.
\end{definition}

\begin{theorem}[Cook-Levin Theorem]
	SAT is NP-complete.
\end{theorem}
\begin{proof}
	It suffices to show that SAT is NP-hard.

	Let $L\in \NP$. By definition, there exists a polynomial-time verifier $M$ for $L$. We want to describe a polynomial-time transformation $x\to \varphi_x$ such that
	\[
		\varphi_x \in \text{SAT} \iff \exists u \in \{0,1\}^{p(|x|)} \text{ such that } M(x,u) = 1\]

	We do so by constructing a tableau of the computation of $M$ on input $(x,u)$, where $u$ is a string of length $p(|x|)$. 
	The tableau is a two-dimensional array where each row represents a configuration of the Turing machine at a specific time step, and each column represents a tape cell. 

	The first row corresponds to the initial configuration. For example, for a starting state $q_0$ and input tape containing $x$ followed by $u$, the first row would look like:

	\[
		\bot \underbrace{q_0}_{\text{state}} \underbrace{w_1 w_2 \ldots w_n}_{\text{input tape}} \underbrace{\# \# \ldots}_{\text{blank cells}}\bot
	\]

	Checking the first row is easy, as we can hardcode the initial configuration into the formula. Name such a formula $\varphi_{\text{start}}$.

	Then after some state transitions, the intermediate rows may look like:

	\[
	\bot \underbrace{\# \# \ldots}_{\text{blank cells}} \underbrace{y_1 y_2 \ldots y_i}_{\text{tape contents}} \underbrace{q_j}_{\text{state}} \underbrace{y_{i+1}\ldots y_m}_{\text{tape contents}} \underbrace{\# \# \ldots}_{\text{blank cells}}\bot
	\]
	We can't be sure about the contents of the tape, but we can check that the transition from one row to the next is valid according to the transition function of $M$. For example, describing a transition on state $q_i$ only concerns a $2 \times 3$ window of the tableau:

	\[
		\begin{array}{ccc}
			\# & q_i & 0 \\
			q_j & \# & 1
		\end{array}
	\]
	Then it suffices to encode the transition rules of $M$ into a windows, and check that every $2 \times 3$ window of the tableau satisfies one of the transition rules. This can be done by a formula $\varphi_{\text{move}}$.

	Also, we want to check that all cells are valid (i.e. there're no multiple characters in a cell) We encode such constraints into a formula $\varphi_{\text{cell}}$.

	Finally, there has to be a $q_{\text{accept}}$ in the last row of the tableau. Searching for this symbol can be encoded into a formula $\varphi_{\text{accept}}$.

	Now we construct the final formula as $\varphi_x = \varphi_{\text{start}} \wedge \varphi_{\text{move}} \wedge \varphi_{\text{cell}} \wedge \varphi_{\text{accept}}$. This formula is satisfiable if and only if there exists a valid tableau of the computation of $M$ on input $(x,u)$ that leads to an accepting configuration, which is equivalent to saying that there exists a $u$ such that $M(x,u) = 1$.

\end{proof}

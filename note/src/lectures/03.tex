\chapter{Proving NP-hard Problems}

\section{NP-hard and NP-complete}

\begin{definition}[Karp Reduction]
	$L \le_p L'$ if there's a polynomial-time computable function $f$ such that for all $x$, $x \in L \iff f(x) \in L'$.
\end{definition}

\begin{definition}[NP-hard and NP-complete]
	$L$ is \textbf{NP-hard} if for all $L' \in \NP$, $L' \le_p L$. $L$ is \textbf{NP-complete} if $L$ is NP-hard and $L \in \NP$.
\end{definition}

We can find a NP-complete problem by simply looking at the definitions.

\begin{problem}[TMSAT]
	\[\text{TMSAT} = \{\langle M, x, 1^n, 1^t\rangle : \exists u\in \{0,1\}^n \text{ such that } M(x,u) \text{ accepts in } t \text{ steps}\}.\]
\end{problem}
\begin{theorem}
	TMSAT is NP-complete.
\end{theorem}
\begin{proof}
	First of all we see that TMSAT is in NP as we can give out $u$ and simulate.

	Now we want to prove that TMSAT is NP-hard. Let $L \in \NP$ and let $M$ be a polynomial-time verifier for $L$. Let $p(n)$ be the polynomial bound on the running time of $M$. We can reduce $L$ to TMSAT by mapping $x$ to $\langle M, x, 1^{p(|x|)}, 1^{p(|x|)}\rangle$. This reduction is clearly polynomial-time computable.
	Now we have that $x \in L$ if and only if there exists a $u$ such that $M(x,u)$ accepts in $p(|x|)$ steps, which is equivalent to $\langle M, x, 1^{p(|x|)}, 1^{p(|x|)}\rangle \in \text{TMSAT}$. Therefore, TMSAT is NP-hard.
\end{proof}

This, however, does not give us a very useful NP-complete problem.

\section{Cook-Levin Theorem}

Cook-Levin theorem is about an interesting NP-complete problem called SAT.

\begin{definition}[CNF and SAT]
	A formula is in \textbf{conjunctive normal form} (CNF) if it is a Boolean formula over variables $u_1, u_2, \ldots, u_n$ that has the form
	\[ \bigwedge_i\left( \bigvee_j v_{ij}\right)\]
	where each $v_{ij}$ is either a variable $u_k$ or its negation $\neg u_k$ (also written as $\overline{u_k}$). A formula is \textbf{satisfiable} if there exists an assignment of truth values to the variables that makes the formula evaluate to true. The \textbf{SAT} problem is the decision problem of determining whether a given CNF formula is satisfiable.
\end{definition}

\begin{theorem}[Cook-Levin Theorem]
	SAT is NP-complete.
\end{theorem}
\begin{proof}
	It suffices to show that SAT is NP-hard.

	Let $L\in \NP$. By definition, there exists a polynomial-time verifier $M$ for $L$. We want to describe a polynomial-time transformation $x\to \varphi_x$ such that
	\[
		\varphi_x \in \text{SAT} \iff \exists u \in \{0,1\}^{p(|x|)} \text{ such that } M(x,u) = 1\]
\end{proof}
